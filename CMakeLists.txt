cmake_minimum_required(VERSION 3.20)
project(TorrentPlayer VERSION 0.1.0 LANGUAGES C CXX)
# Milestone 1: engine / SegmentStore / scheduler core (C++17). Builds with Command Line Tools + Homebrew (no Xcode).
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
add_compile_options(-Wall -Wextra -Wno-unused-parameter)

find_package(PkgConfig REQUIRED)
set(ENV{PKG_CONFIG_PATH} "/opt/homebrew/lib/pkgconfig:/opt/homebrew/opt/openssl@3/lib/pkgconfig:$ENV{PKG_CONFIG_PATH}")
pkg_check_modules(LIBTORRENT REQUIRED IMPORTED_TARGET libtorrent-rasterbar)
find_package(Threads REQUIRED)

# ---- tpcore: pure logic + store + engine
add_library(tpcore STATIC
  core/src/config.cpp
  core/src/window_planner.cpp
  core/src/disk_policy.cpp
  core/src/web_seed_policy.cpp
  core/src/segment_store.cpp
  core/src/diagnostics.cpp
  core/src/audit.cpp
  core/src/engine.cpp
  core/src/stream_session.cpp
  core/src/tp_c.cpp)
target_include_directories(tpcore PUBLIC core/include)
target_include_directories(tpcore SYSTEM PUBLIC /opt/homebrew/include /opt/homebrew/opt/openssl@3/include)
target_link_directories(tpcore PUBLIC /opt/homebrew/lib /opt/homebrew/opt/openssl@3/lib)
target_link_libraries(tpcore PUBLIC PkgConfig::LIBTORRENT Threads::Threads)

# ---- tests (plain executables run by CTest; no external test framework)
enable_testing()
function(tp_test name src)
  add_executable(${name} ${src})
  target_include_directories(${name} PRIVATE tests)
  target_link_libraries(${name} PRIVATE tpcore)
  add_test(NAME ${name} COMMAND ${name})
endfunction()
file(GLOB UNIT_TESTS tests/unit/*.cpp)
foreach(f ${UNIT_TESTS})
  get_filename_component(n ${f} NAME_WE)
  tp_test(${n} ${f})
  set_tests_properties(${n} PROPERTIES LABELS "unit;mandatory-gate")
endforeach()
file(GLOB TOOL_SRCS tools/*.cpp)
foreach(f ${TOOL_SRCS})
  get_filename_component(n ${f} NAME_WE)
  add_executable(${n} ${f})
  target_link_libraries(${n} PRIVATE tpcore)
endforeach()

# ---- integration tests (loopback swarm; mandatory gates M1-G2..G8). Run with: ctest -L integration
file(GLOB IT_SRCS tests/integration/it_*.cpp)
foreach(f ${IT_SRCS})
  get_filename_component(n ${f} NAME_WE)
  add_executable(${n} ${f})
  target_include_directories(${n} PRIVATE tests tests/integration)
  target_compile_definitions(${n} PRIVATE TP_BUILD_DIR="${CMAKE_BINARY_DIR}")
  target_link_libraries(${n} PRIVATE tpcore)
endforeach()
